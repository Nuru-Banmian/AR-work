async (page) => {
  const check = (condition, message) => { if (!condition) throw new Error(message) }
  await page.getByRole('button', {name:'全部 3',exact:true}).waitFor()
  await page.getByRole('button',{name:'清除筛选',exact:true}).click()
  await page.getByRole('searchbox').fill('北区')
  check(await page.getByRole('button',{name:/^查看 /}).count() === 2,'description search')
  await page.getByRole('button',{name:'待处理 1',exact:true}).click()
  await page.getByRole('combobox',{name:'优先级筛选',exact:true}).selectOption('high')
  check(await page.getByRole('button',{name:'查看 阀门漏水',exact:true}).count() === 1,'combined filters')
  await page.getByRole('combobox',{name:'优先级筛选',exact:true}).selectOption('low')
  await page.getByText('没有匹配的问题。请修改关键词或清除筛选。').waitFor()
  check(await page.getByRole('button',{name:'全部 3',exact:true}).count() === 1,'statistics remain global')
  await page.getByRole('button',{name:'清除筛选',exact:true}).click()
  await page.getByRole('combobox',{name:'排序',exact:true}).selectOption('priority')
  check((await page.getByRole('button',{name:/^查看 /}).allTextContents()).join(',') === '阀门漏水,照明故障,护栏修复','priority ordering')
  await page.getByRole('button',{name:'查看 阀门漏水',exact:true}).click()
  check(await page.getByRole('button',{name:'关闭详情',exact:true}).evaluate(el=>el===document.activeElement),'detail initial focus')
  await page.getByRole('button',{name:'编辑内容',exact:true}).click()
  await page.getByLabel('阀门漏水 标题',{exact:true}).fill('阀门漏水 草稿')
  await page.getByRole('button',{name:'查看 照明故障',exact:true}).click()
  await page.getByRole('button',{name:'查看 阀门漏水',exact:true}).click()
  check(await page.getByLabel('阀门漏水 标题',{exact:true}).inputValue() === '阀门漏水 草稿','switch preserves draft')
  await page.getByRole('searchbox').fill('照明')
  await page.getByText('当前问题不符合筛选，详情和草稿仍保留。').waitFor()
  check(await page.getByLabel('阀门漏水 标题',{exact:true}).inputValue() === '阀门漏水 草稿','filter preserves draft')
  await page.getByRole('button',{name:'清除筛选',exact:true}).click()
  const list = await (await page.request.get('http://127.0.0.1:5197/api/issues')).json()
  const issue = list.data.find(item=>item.title==='阀门漏水')
  await page.request.patch(`http://127.0.0.1:5197/api/issues/${issue.id}/status`,{data:{status:'in_progress'}})
  await page.getByLabel('阀门漏水 处理状态',{exact:true}).selectOption('resolved')
  await page.getByRole('button',{name:'已解决 2',exact:true}).waitFor()
  check(await page.getByLabel('阀门漏水 标题',{exact:true}).inputValue() === '阀门漏水 草稿','SSE preserves draft')
  await page.getByRole('button',{name:'保存内容',exact:true}).click()
  await page.getByRole('heading',{name:'阀门漏水 草稿',exact:true}).waitFor()
  check((await (await page.request.get('http://127.0.0.1:5197/api/issues')).json()).data.find(item=>item.id===issue.id).status === 'resolved','content save preserves status')
  await page.getByRole('button',{name:'阀门漏水 草稿 查看现场大图',exact:true}).click()
  await page.getByRole('dialog',{name:'现场大图',exact:true}).waitFor()
  await page.keyboard.press('Escape')
  check(await page.getByRole('dialog').count() === 0,'Escape closes photo only')
  check(await page.getByRole('button',{name:'阀门漏水 草稿 查看现场大图',exact:true}).evaluate(el=>el===document.activeElement),'photo restores focus')
  await page.keyboard.press('Escape')
  check(await page.getByLabel('问题详情',{exact:true}).count() === 0,'Escape closes detail')
  await page.getByRole('button',{name:'查看 阀门漏水 草稿',exact:true}).click()
  await page.getByRole('button',{name:'三维模型',exact:true}).click()
  await page.locator('canvas').waitFor()
  await page.getByRole('button',{name:'显示/隐藏标记',exact:true}).click()
  await page.getByRole('button',{name:'显示/隐藏颜色',exact:true}).click()
  await page.getByRole('button',{name:'复位模型',exact:true}).click()
  await page.getByRole('button',{name:'查看 照明故障',exact:true}).click()
  check(await page.locator('canvas').count() === 0,'new detail defaults to photo and disposes model')
  await page.getByRole('button',{name:'现场照片',exact:true}).waitFor()
  console.log('PASS filters, global counts, priority sort, draft switch/filter/SSE, status/content, photo modal focus, model integration')
}
