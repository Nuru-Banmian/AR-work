async (page) => {
 const check=(condition,message)=>{if(!condition)throw new Error(message)}
 await page.getByRole('button',{name:'查看 照明故障',exact:true}).click()
 await page.getByRole('button',{name:'编辑内容',exact:true}).click()
 await page.getByLabel('照明故障 标题',{exact:true}).fill('保存失败保留草稿')
 let release
 const gate=new Promise(resolve=>{release=resolve})
 let started
 const requested=new Promise(resolve=>{started=resolve})
 await page.route('**/content',async route=>{started();await gate;await route.fulfill({status:500,contentType:'application/json',body:'{}'})})
 await page.getByRole('button',{name:'保存内容',exact:true}).click()
 await requested
 await page.getByRole('button',{name:'查看 护栏修复',exact:true}).click()
 await page.getByRole('button',{name:'查看 照明故障',exact:true}).click()
 check(await page.getByRole('button',{name:'保存内容',exact:true}).isDisabled(),'pending operation remains locked across remount')
 check(await page.getByLabel('照明故障 标题',{exact:true}).isDisabled(),'pending draft remains locked')
 release()
 await page.getByRole('button',{name:'保存内容',exact:true}).waitFor({state:'visible'})
 await page.waitForFunction(()=>!Array.from(document.querySelectorAll('button')).find(button=>button.textContent==='保存内容')?.disabled)
 await page.unroute('**/content')
 check(await page.getByLabel('照明故障 标题',{exact:true}).inputValue()==='保存失败保留草稿','failed pending save retains draft')
 for(const width of [1024,1366,1920]) {
  await page.setViewportSize({width,height:900})
  check(await page.evaluate(()=>document.documentElement.scrollWidth<=innerWidth),'no page overflow at '+width)
  check(await page.getByRole('button',{name:'关闭详情',exact:true}).isVisible(),'detail accessible at '+width)
  await page.getByLabel('照明故障 标题',{exact:true}).focus()
  await page.keyboard.press('Tab')
  check(await page.getByLabel('照明故障 描述',{exact:true}).evaluate(el=>el===document.activeElement),'editor tab order at '+width)
 }
 await page.getByRole('button',{name:'取消编辑',exact:true}).click()
 await page.getByRole('button',{name:'永久删除',exact:true}).click()
 for(let i=0;i<8;i++) {
  await page.keyboard.press('Tab')
  check(await page.evaluate(()=>!!document.activeElement.closest('dialog')),'modal traps keyboard')
 }
 await page.keyboard.press('Escape')
 console.log('PASS pending save isolation/remount lock/failure draft; 1024/1366/1920 layout and editor keyboard; modal focus trap')
}
